import { MetadataRoute } from 'next'

export default function robots(): MetadataRoute.Robots {
  return {
    rules: [
      {
        userAgent: '*',
        allow: '/',
        disallow: [
          '/_next/static/',   // block font files, JS chunks, CSS — not useful to Google
          '/_next/image/',    // block Next.js image optimizer URLs
          '/api/',            // block API routes
          '/admin/',          // block any admin panel
        ],
      },
    ],
    sitemap: 'https://www.pelicantravelsandtours.com/sitemap.xml',
  }
}
